1//! Reader → writer messages and lane routing.
2
3use crossbeam_channel::{Receiver, Sender};
4
5use super::buffer::PooledBuf;
6use crate::domain::job::FileId;
7use crate::engine::error::EngineError;
8
9/// What travels from readers to writers. Borrows its buffer from the pool `'p`.
10#[derive(Debug)]
11pub enum Message<'p> {
12    /// `data` belongs at `offset` in file `file`.
13    Chunk {
14        /// Which file.
15        file: FileId,
16        /// Byte offset within the file.
17        offset: u64,
18        /// The bytes.
19        data: PooledBuf<'p>,
20    },
21    /// The reader finished `file`; `len` bytes were read in total.
22    End {
23        /// Which file.
24        file: FileId,
25        /// Final length (may differ from the scanned size if the file changed).
26        len: u64,
27    },
28}
29
30impl Message<'_> {
31    /// The file this message concerns.
32    pub const fn file(&self) -> FileId {
33        match self {
34            Self::Chunk { file, .. } | Self::End { file, .. } => *file,
35        }
36    }
37}
38
39/// Sends each file's messages to a fixed writer lane.
40///
41/// Because a file always goes to the same lane and has exactly one reader,
42/// its `End` is always received after its last `Chunk`.
43#[derive(Debug, Clone)]
44pub struct Router<'p> {
45    lanes: Vec<Sender<Message<'p>>>,
46}
47
48impl<'p> Router<'p> {
49    /// The lane index for `file`.
50    pub fn lane_for(&self, file: FileId) -> usize {
51        file.index() % self.lanes.len()
52    }
53
54    /// Routes `msg` to its lane. Fails with `Cancelled` if that writer is gone.
55    pub fn send(&self, msg: Message<'p>) -> Result<(), EngineError> {
56        self.lanes[self.lane_for(msg.file())]
57            .send(msg)
58            .map_err(|_| EngineError::Cancelled)
59    }
60}
61
62/// Creates `writers` bounded lanes of `depth` messages each.
63pub fn lanes<'p>(writers: usize, depth: usize) -> (Router<'p>, Vec<Receiver<Message<'p>>>) {
64    let (senders, receivers) = (0..writers.max(1))
65        .map(|_| crossbeam_channel::bounded(depth))
66        .unzip();
67    (Router { lanes: senders }, receivers)
68}

70#[cfg(test)]
71mod tests {
72    use super::*;
73
74    #[test]
75    fn router_sends_same_file_to_same_lane() {
76        let (router, _rx) = lanes(3, 4);
77        let file = FileId::new(7);
78        assert_eq!(router.lane_for(file), router.lane_for(file));
79    }
80
81    #[test]
82    fn router_spreads_files_over_lanes() {
83        let (router, _rx) = lanes(2, 4);
84        assert_ne!(
85            router.lane_for(FileId::new(0)),
86            router.lane_for(FileId::new(1))
87        );
88    }
89
90    #[test]
91    fn send_to_dropped_writer_is_cancelled() {
92        let (router, receivers) = lanes(1, 4);
93        drop(receivers);
94        let msg = Message::End {
95            file: FileId::new(0),
96            len: 0,
97        };
98        assert!(matches!(router.send(msg), Err(EngineError::Cancelled)));
99    }
