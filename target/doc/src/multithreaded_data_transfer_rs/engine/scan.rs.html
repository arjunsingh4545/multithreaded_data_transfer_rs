15/// Report scan progress every this many files.
16const REPORT_EVERY: usize = 256;
17
18/// Shared, read-only walk state. Borrowed by every rayon task.
19struct ScanCtx<'a, P> {
20    src_root: &'a Path,
21    dst_root: &'a Path,
22    progress: &'a P,
23    files: AtomicUsize,
24    bytes: AtomicU64,
25}
26
27/// One directory's worth of results; merged up the tree.
28#[derive(Default)]
29struct Partial {
30    dirs: Vec<PathBuf>,
31    symlinks: Vec<SymlinkJob>,
32    files: Vec<FileJob>,
33}
34
35impl Partial {
36    fn merge(mut self, mut other: Self) -> Self {
37        self.dirs.append(&mut other.dirs);
38        self.symlinks.append(&mut other.symlinks);
39        self.files.append(&mut other.files);
40        self
41    }
42}
43
44/// Walks `src_root`, mapping every entry to its place under `dst_root`.
45///
46/// Fails fast on the first unreadable directory or entry. Entries other than
47/// directories, regular files and symlinks (sockets, FIFOs, devices) are skipped.
48pub fn scan_tree<P: ProgressSink>(
49    src_root: &Path,
50    dst_root: &Path,
51    progress: &P,
52) -> Result<ScanManifest, EngineError> {
53    let ctx = ScanCtx {
54        src_root,
55        dst_root,
56        progress,
57        files: AtomicUsize::new(0),
58        bytes: AtomicU64::new(0),
59    };
60
61    let found = walk(&ctx, src_root)?;
62    let mut directories = Vec::with_capacity(found.dirs.len() + 1);
63    directories.push(dst_root.to_path_buf());
64    directories.extend(found.dirs);
65    // Parents before children so creation never needs to recurse.
66    directories.sort_by_key(|p| p.components().count());
67
68    let total_bytes = ctx.bytes.into_inner();
69    progress.scanned(found.files.len(), total_bytes);
70    Ok(ScanManifest {
71        directories,
72        symlinks: found.symlinks,
73        files: found.files,
74        total_bytes,
75    })
76}

78fn walk<P: ProgressSink>(ctx: &ScanCtx<'_, P>, dir: &Path) -> Result<Partial, EngineError> {
79    let dir_err = |source| EngineError::DirAccess {
80        path: dir.to_path_buf(),
81        source,
82    };
83    let mut entries: Vec<DirEntry> = fs::read_dir(dir)
84        .map_err(dir_err)?
85        .collect::<Result<_, _>>()
86        .map_err(dir_err)?;
87    // Inode order makes the following `stat` calls far less seeky on HDDs.
88    entries.sort_unstable_by_key(DirEntryExt::ino);
89
90    let mut here = Partial::default();
91    let mut subdirs = Vec::new();
92    for entry in entries {
93        let src = entry.path();
94        let meta = fs::symlink_metadata(&src).map_err(|source| EngineError::Metadata {
95            path: src.clone(),
96            source,
97        })?;
98        let Some(dst) = rebase(&src, ctx.src_root, ctx.dst_root) else {
99            continue;
100        };
101        let kind = meta.file_type();
102
103        if kind.is_symlink() {
104            let target = fs::read_link(&src).map_err(|source| EngineError::Symlink {
105                path: src.clone(),
106                source,
107            })?;
108            here.symlinks.push(SymlinkJob { src, dst, target });
109        } else if kind.is_dir() {
110            here.dirs.push(dst);
111            subdirs.push(src);
112        } else if kind.is_file() {
113            ctx.record(meta.len());
114            here.files.push(FileJob {
115                size: meta.len(),
116                ino: meta.ino(),
117                meta: FileMeta::from(&meta),
118                src,
119                dst,
120            });
121        }
122    }
123
124    let below = subdirs
125        .par_iter()
126        .map(|sub| walk(ctx, sub))
127        .try_reduce(Partial::default, |a, b| Ok(a.merge(b)))?;
128    Ok(here.merge(below))
129}
130
131impl<P: ProgressSink> ScanCtx<'_, P> {
132    fn record(&self, size: u64) {
133        let files = self.files.fetch_add(1, Ordering::Relaxed) + 1;
134        let bytes = self.bytes.fetch_add(size, Ordering::Relaxed) + size;
135        if files.is_multiple_of(REPORT_EVERY) {
136            self.progress.scanned(files, bytes);
137        }
138    }
139}

141#[cfg(test)]
142mod tests {
143    use super::*;
144    use crate::engine::progress::NoProgress;
145    use std::os::unix::fs::symlink;
146
147    fn tree() -> tempfile::TempDir {
148        let root = tempfile::tempdir().unwrap();
149        let p = root.path();
150        fs::create_dir_all(p.join("a/b")).unwrap();
151        fs::create_dir_all(p.join("empty")).unwrap();
152        fs::write(p.join("top.txt"), b"12345").unwrap();
153        fs::write(p.join("a/b/deep.bin"), vec![0u8; 100]).unwrap();
154        symlink("top.txt", p.join("link")).unwrap();
155        root
156    }
157
158    #[test]
159    fn finds_all_files_and_bytes() {
160        let src = tree();
161        let manifest = scan_tree(src.path(), Path::new("/out"), &NoProgress).unwrap();
162        assert_eq!((manifest.files.len(), manifest.total_bytes), (2, 105));
163    }
164
165    #[test]
166    fn records_symlinks_without_following() {
167        let src = tree();
168        let manifest = scan_tree(src.path(), Path::new("/out"), &NoProgress).unwrap();
169        assert_eq!(manifest.symlinks[0].target, PathBuf::from("top.txt"));
170    }
171
172    #[test]
173    fn maps_paths_under_destination() {
174        let src = tree();
175        let manifest = scan_tree(src.path(), Path::new("/out"), &NoProgress).unwrap();
176        let deep = manifest.files.iter().find(|f| f.size == 100).unwrap();
177        assert_eq!(deep.dst, PathBuf::from("/out/a/b/deep.bin"));
178    }
179
180    #[test]
181    fn directories_include_root_and_empty_dirs_parents_first() {
182        let src = tree();
183        let manifest = scan_tree(src.path(), Path::new("/out"), &NoProgress).unwrap();
184        assert_eq!(manifest.directories[0], PathBuf::from("/out"));
185        assert!(manifest.directories.contains(&PathBuf::from("/out/empty")));
186    }
187
188    #[test]
189    fn missing_source_is_dir_access_error() {
190        let result = scan_tree(
191            Path::new("/definitely/missing"),
192            Path::new("/out"),
193            &NoProgress,
194        );
195        assert!(matches!(result, Err(EngineError::DirAccess { .. })));
196    }
