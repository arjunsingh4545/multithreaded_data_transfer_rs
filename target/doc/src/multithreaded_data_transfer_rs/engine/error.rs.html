12/// Something went wrong while scanning or copying.
13#[derive(Debug, Error)]
14pub enum EngineError {
15    /// The source path does not exist.
16    #[error("source '{0}' does not exist")]
17    SourceNotFound(PathBuf),
18    /// The source path is not a directory.
19    #[error("source '{0}' is not a directory")]
20    SourceNotDir(PathBuf),
21    /// The destination path could not be resolved.
22    #[error("cannot resolve destination '{path}': {source}")]
23    DestUnresolvable {
24        /// Destination as given.
25        path: PathBuf,
26        /// Underlying error.
27        #[source]
28        source: io::Error,
29    },
30    /// The destination is the source or lies inside it.
31    #[error("destination '{dst}' is inside source '{src}'; this would copy the tree into itself")]
32    DestInsideSource {
33        /// Source as given.
34        src: PathBuf,
35        /// Destination as given.
36        dst: PathBuf,
37    },
38    /// `--if-exists fail` found an existing file.
39    #[error("'{0}' already exists (policy: fail)")]
40    DestExists(PathBuf),
41    /// A source directory could not be listed.
42    #[error("cannot read directory '{path}': {source}")]
43    DirAccess {
44        /// Directory path.
45        path: PathBuf,
46        /// Underlying error.
47        #[source]
48        source: io::Error,
49    },
50    /// An entry's metadata could not be read.
51    #[error("cannot inspect '{path}': {source}")]
52    Metadata {
53        /// Entry path.
54        path: PathBuf,
55        /// Underlying error.
56        #[source]
57        source: io::Error,
58    },
59    /// A symlink could not be read or created.
60    #[error("cannot copy symlink '{path}': {source}")]
61    Symlink {
62        /// Link path.
63        path: PathBuf,
64        /// Underlying error.
65        #[source]
66        source: io::Error,
67    },
68    /// A destination directory could not be created.
69    #[error("cannot create directory '{path}': {source}")]
70    DirCreate {
71        /// Directory path.
72        path: PathBuf,
73        /// Underlying error.
74        #[source]
75        source: io::Error,
76    },
77    /// Copying a file's contents failed.
78    #[error("cannot copy '{src}' → '{dst}': {source}")]
79    FileCopy {
80        /// Source file.
81        src: PathBuf,
82        /// Destination file.
83        dst: PathBuf,
84        /// Underlying error.
85        #[source]
86        source: io::Error,
87    },
88    /// A worker thread could not be started.
89    #[error("cannot start worker thread: {0}")]
90    ThreadSpawn(#[source] io::Error),
91    /// A worker thread panicked.
92    #[error("a {role} thread panicked")]
93    WorkerPanicked {
94        /// Which kind of worker.
95        role: Role,
96    },
97    /// The copy was interrupted (Ctrl-C or a failure elsewhere).
98    #[error("operation cancelled")]
99    Cancelled,
100}
101
102impl EngineError {
103    /// An error constructor for I/O failures while copying `job`.
104    ///
105    /// The returned closure borrows `job`, so it can be passed straight to `map_err`.
106    pub fn copy_failed(job: &FileJob) -> impl Fn(io::Error) -> Self + '_ {
107        move |source| Self::FileCopy {
108            src: job.src.clone(),
109            dst: job.dst.clone(),
110            source,
111        }
112    }
113}
114
115impl Cancellation for EngineError {
116    fn is_cancellation(&self) -> bool {
117        matches!(self, Self::Cancelled)
118    }
