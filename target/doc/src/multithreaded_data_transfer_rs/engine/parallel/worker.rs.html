1//! One parallel-mode worker: pulls whole files off the queue and copies them.
2
3use std::fs::{self, File};
4use std::io::{Read, Write};
5
6use tokio_util::sync::CancellationToken;
7
8use crate::domain::job::FileJob;
9use crate::engine::error::EngineError;
10use crate::engine::finalize;
11use crate::engine::progress::{ProgressSink, Role, WorkerIdx};
12use crate::engine::queue::JobQueue;
13use crate::engine::stats::CopyStats;
14use crate::util::fs::{RemoveOnDrop, display_name};
15
16/// Below this size the kernel copy path (`copy_file_range`) is used directly.
17const SMALL_FILE: u64 = 1 << 20;
18
19/// Borrowed state for one worker thread.
20pub struct Worker<'s, P> {
21    /// Position among copiers.
22    pub idx: WorkerIdx,
23    /// Shared job cursor.
24    pub queue: &'s JobQueue<'s>,
25    /// Buffer size for streamed files.
26    pub chunk: usize,
27    /// Progress sink.
28    pub progress: &'s P,
29    /// Stop signal.
30    pub cancel: &'s CancellationToken,
31}
32
33impl<P: ProgressSink> Worker<'_, P> {
34    /// Copies jobs until the queue is drained. Cancels everyone else on failure.
35    pub fn run(self) -> Result<CopyStats, EngineError> {
36        let result = self.drain();
37        if result.is_err() {
38            self.cancel.cancel();
39        }
40        self.progress.worker_done(Role::Copier, self.idx);
41        result
42    }
43
44    fn drain(&self) -> Result<CopyStats, EngineError> {
45        let mut buffer = vec![0u8; self.chunk];
46        let mut stats = CopyStats::default();
47        while let Some((_, job)) = self.queue.next() {
48            if self.cancel.is_cancelled() {
49                return Err(EngineError::Cancelled);
50            }
51            self.progress
52                .file_started(Role::Copier, self.idx, &display_name(&job.src), job.size);
53            let metadata_ok = if job.size < SMALL_FILE {
54                self.copy_small(job)?
55            } else {
56                self.copy_streamed(job, &mut buffer)?
57            };
58            stats.record_file(job.size, metadata_ok);
59        }
60        Ok(stats)
61    }
62
63    /// Kernel-side copy; returns whether metadata could be restored.
64    fn copy_small(&self, job: &FileJob) -> Result<bool, EngineError> {
65        let mut guard = RemoveOnDrop::new(&job.dst);
66        let copied = fs::copy(&job.src, &job.dst).map_err(EngineError::copy_failed(job))?;
67        guard.disarm();
68        self.progress.advanced(Role::Copier, self.idx, copied);
69        Ok(finalize::apply_to_path(&job.dst, &job.meta).is_ok())
70    }
71
72    /// Chunked copy with per-chunk progress and cancellation checks.
73    fn copy_streamed(&self, job: &FileJob, buffer: &mut [u8]) -> Result<bool, EngineError> {
74        let fail = EngineError::copy_failed(job);
75        let mut reader = File::open(&job.src).map_err(&fail)?;
76        let mut guard = RemoveOnDrop::new(&job.dst);
77        let mut writer = File::create(&job.dst).map_err(&fail)?;
78
79        loop {
80            if self.cancel.is_cancelled() {
81                return Err(EngineError::Cancelled);
82            }
83            let n = match reader.read(buffer) {
84                Ok(0) => break,
85                Ok(n) => n,
86                Err(e) if e.kind() == std::io::ErrorKind::Interrupted => continue,
87                Err(e) => return Err(fail(e)),
88            };
89            writer.write_all(&buffer[..n]).map_err(&fail)?;
90            self.progress.advanced(Role::Copier, self.idx, n as u64);
91        }
92
93        guard.disarm();
94        Ok(finalize::apply_to_file(&writer, &job.meta).is_ok())
95    }
96}
