1//! `probe`: show the detected disk behind each path.
2
3use std::num::NonZeroUsize;
4use std::process::ExitCode;
5
6use super::{Context, Run};
7use crate::cli::args::ProbeArgs;
8use crate::cli::report;
9use crate::domain::plan::{Tuning, WorkloadStats, plan};
10use crate::domain::topology::{KindOverrides, Topology};
11use crate::engine::error::EngineError;
12use crate::error::Result;
13use crate::platform::device::{DeviceProbe, PlatformProbe};
14use crate::util::fs::{canonicalize_partial, nearest_existing};
15
16impl Run for ProbeArgs {
17    fn run(self, ctx: &Context) -> Result<ExitCode> {
18        let probe = PlatformProbe::default();
19        let disks = self
20            .paths
21            .iter()
22            .map(|path| {
23                let resolved =
24                    canonicalize_partial(path).map_err(|source| EngineError::DestUnresolvable {
25                        path: path.clone(),
26                        source,
27                    })?;
28                let existing = nearest_existing(&resolved).unwrap_or(&resolved);
29                let disk = probe.probe(existing)?;
30                report::probe_row(&ctx.out, path, &disk);
31                Ok(disk)
32            })
33            .collect::<Result<Vec<_>>>()?;
34
35        if let [src, dst] = disks.as_slice() {
36            let topology = Topology::new(src.clone(), dst.clone(), KindOverrides::default());
37            let cores = std::thread::available_parallelism().unwrap_or(NonZeroUsize::MIN);
38            let plan = plan(
39                WorkloadStats::default(),
40                topology.transfer(),
41                &Tuning::default(),
42                cores,
43            );
44            ctx.out.field("copy", plan);
45            ctx.out.detail(format!("why: {}", plan.reason));
46        }
47        Ok(ExitCode::SUCCESS)
48    }
49}
