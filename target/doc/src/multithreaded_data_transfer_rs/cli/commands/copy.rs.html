18impl Run for CopyArgs {
19    fn run(self, ctx: &Context) -> Result<ExitCode> {
20        let out = &ctx.out;
21        let flags = FlagsUsed::from(&self);
22        let (dry_run, assume_yes) = (self.safety.dry_run, self.safety.yes);
23        let request = CopyRequest::from(self);
24
25        if !dry_run && !assume_yes && !confirm_destination(*out, &request) {
26            out.warn("aborted; nothing was copied");
27            return Ok(ExitCode::FAILURE);
28        }
29
30        let engine = Engine::new();
31        let prepared = {
32            let spinner = ScanSpinner::new(&request.src, out.shows_progress());
33            let prepared = engine.prepare(&request, &spinner);
34            spinner.finish();
35            prepared?
36        };
37
38        report::plan(out, &prepared);
39        if let Some(hint) = flags.hint_for(&prepared.plan().strategy) {
40            out.hint(hint);
41        }
42        if dry_run {
43            out.success("dry run: nothing was written");
44            return Ok(ExitCode::SUCCESS);
45        }
46
47        let summary = if out.shows_progress() {
48            let ui = CopyUi::new(
49                prepared.manifest().total_bytes,
50                UiLayout::from(&prepared.plan().strategy),
51            );
52            let result = execute(&engine, &prepared, &ui, ctx);
53            ui.finish(result.is_ok());
54            result?
55        } else {
56            execute(&engine, &prepared, &NoProgress, ctx)?
57        };
58
59        report::summary(out, &summary);
60        Ok(ExitCode::SUCCESS)
61    }
62}

64/// Statically dispatched over the progress sink (bars or nothing).
65fn execute<P: ProgressSink>(
66    engine: &Engine,
67    prepared: &Prepared,
68    progress: &P,
69    ctx: &Context,
70) -> Result<Summary> {
71    engine.execute(prepared, progress, &ctx.cancel)
72}

74/// Asks before copying into a non-empty destination with `--if-exists overwrite`.
75///
76/// Never blocks scripts: without an interactive terminal the answer is "yes".
77fn confirm_destination(out: Output, request: &CopyRequest) -> bool {
78    if request.existing != ExistingPolicy::Overwrite
79        || !out.is_interactive()
80        || !is_non_empty_dir(&request.dst)
81    {
82        return true;
83    }
84    eprint!(
85        "{} '{}' is not empty; files with the same names will be overwritten. Continue? [y/N] ",
86        console::style("?").cyan().bold(),
87        request.dst.display()
88    );
89    io::stderr().flush().ok();
90    let mut answer = String::new();
91    io::stdin().lock().read_line(&mut answer).is_ok()
92        && matches!(answer.trim(), "y" | "Y" | "yes" | "YES")
93}

95fn is_non_empty_dir(path: &Path) -> bool {
96    fs::read_dir(path).is_ok_and(|mut entries| entries.next().is_some())
