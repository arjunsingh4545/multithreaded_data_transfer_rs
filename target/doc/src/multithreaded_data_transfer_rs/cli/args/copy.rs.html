1//! Arguments of the `copy` command, grouped by help heading.
2
3use std::num::NonZeroUsize;
4use std::path::PathBuf;
5
6use clap::{Args, ValueHint};
7
8use super::value::{ExistingArg, KindArg, ModeArg, OrderArg};
9use crate::util::units::ByteSize;
10
11/// `copy <SRC> <DST> [OPTIONS]`
12#[derive(Debug, Clone, Args)]
13pub struct CopyArgs {
14    /// Source and destination.
15    #[command(flatten)]
16    pub paths: CopyPaths,
17
18    /// Safety options.
19    #[command(flatten)]
20    pub safety: SafetyArgs,
21
22    /// Performance options.
23    #[command(flatten)]
24    pub perf: PerfArgs,
25
26    /// Disk detection overrides.
27    #[command(flatten)]
28    pub detect: DetectArgs,
29}
30
31/// The two positional paths.
32#[derive(Debug, Clone, Args)]
33pub struct CopyPaths {
34    /// Directory to copy from
35    #[arg(value_name = "SRC", value_hint = ValueHint::DirPath)]
36    pub src: PathBuf,
37
38    /// Directory to copy into (created if missing)
39    #[arg(value_name = "DST", value_hint = ValueHint::DirPath)]
40    pub dst: PathBuf,
41}
42
43/// What may be touched.
44#[derive(Debug, Clone, Args)]
45#[command(next_help_heading = "Safety")]
46pub struct SafetyArgs {
47    /// Scan and show the plan without writing anything
48    #[arg(short = 'n', long)]
49    pub dry_run: bool,
50
51    /// What to do with files that already exist at DST
52    #[arg(long, value_name = "POLICY", value_enum, default_value_t)]
53    pub if_exists: ExistingArg,
54
55    /// Don't ask before copying into a non-empty DST
56    #[arg(short = 'y', long)]
57    pub yes: bool,
58}
59
60/// Performance knobs; leave unset to let the planner decide.
61#[derive(Debug, Clone, Args)]
62#[command(next_help_heading = "Performance")]
63pub struct PerfArgs {
64    /// Copy strategy
65    #[arg(short, long, value_enum, default_value_t)]
66    pub mode: ModeArg,
67
68    /// Concurrent copies in parallel mode
69    #[arg(short = 'j', long, value_name = "N")]
70    pub workers: Option<NonZeroUsize>,
71
72    /// Reader threads in pipeline mode
73    #[arg(long, value_name = "N")]
74    pub readers: Option<NonZeroUsize>,
75
76    /// Writer threads in pipeline mode
77    #[arg(long, value_name = "N")]
78    pub writers: Option<NonZeroUsize>,
79
80    /// I/O size per request, e.g. 512K, 8M
81    #[arg(long, value_name = "SIZE", env = "FASTCP_CHUNK")]
82    pub chunk_size: Option<ByteSize>,
83
84    /// Buffer memory in pipeline mode [default: 1G, 4G for same-disk copies]
85    #[arg(long, value_name = "SIZE", env = "FASTCP_MEM")]
86    pub mem: Option<ByteSize>,
87
88    /// Copy order [default: chosen by plan]
89    #[arg(long, value_enum)]
90    pub order: Option<OrderArg>,
91}
92
93/// Disk detection overrides.
94#[derive(Debug, Clone, Args)]
95#[command(next_help_heading = "Disk detection")]
96pub struct DetectArgs {
97    /// Treat the source disk as this kind instead of detecting it
98    #[arg(long, value_name = "KIND", value_enum)]
99    pub src_disk: Option<KindArg>,
100
101    /// Treat the destination disk as this kind instead of detecting it
102    #[arg(long, value_name = "KIND", value_enum)]
103    pub dst_disk: Option<KindArg>,
104}
