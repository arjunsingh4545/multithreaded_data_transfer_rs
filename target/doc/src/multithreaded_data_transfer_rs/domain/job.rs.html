3use std::fs::{Metadata, Permissions};

8/// Index of a file job inside the (ordered) job slice.
9#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash, PartialOrd, Ord)]
10pub struct FileId(usize);
11
12impl FileId {
13    /// Wraps a slice index.
14    pub const fn new(index: usize) -> Self {
15        Self(index)
16    }
17
18    /// The slice index.
19    pub const fn index(self) -> usize {
20        self.0
21    }
22}
23
24/// Source metadata that is re-applied to the destination after copying.
25#[derive(Debug, Clone, PartialEq, Eq)]
26pub struct FileMeta {
27    /// Mode bits.
28    pub permissions: Permissions,
29    /// Last modification time.
30    pub modified: FileTime,
31    /// Last access time.
32    pub accessed: FileTime,
33}
34
35impl From<&Metadata> for FileMeta {
36    fn from(meta: &Metadata) -> Self {
37        Self {
38            permissions: meta.permissions(),
39            modified: FileTime::from_last_modification_time(meta),
40            accessed: FileTime::from_last_access_time(meta),
41        }
42    }
43}
44
45/// One regular file to copy.
46#[derive(Debug, Clone, PartialEq, Eq)]
47pub struct FileJob {
48    /// Source path.
49    pub src: PathBuf,
50    /// Destination path.
51    pub dst: PathBuf,
52    /// Size in bytes at scan time.
53    pub size: u64,
54    /// Source inode number (used for inode ordering).
55    pub ino: u64,
56    /// Metadata to restore on the destination.
57    pub meta: FileMeta,
58}
59
60/// One symbolic link to recreate (the link itself, never its target).
61#[derive(Debug, Clone, PartialEq, Eq)]
62pub struct SymlinkJob {
63    /// Source link path.
64    pub src: PathBuf,
65    /// Destination link path.
66    pub dst: PathBuf,
67    /// The link's target, stored verbatim.
68    pub target: PathBuf,
69}
70
71/// Everything found under the source root.
72#[derive(Debug, Clone, Default, PartialEq, Eq)]
73pub struct ScanManifest {
74    /// Destination directories, parents before children.
75    pub directories: Vec<PathBuf>,
76    /// Symlinks to recreate.
77    pub symlinks: Vec<SymlinkJob>,
78    /// Regular files to copy.
79    pub files: Vec<FileJob>,
80    /// Sum of all file sizes.
81    pub total_bytes: u64,
82}
83
84impl ScanManifest {
85    /// Recomputes [`total_bytes`](Self::total_bytes) after files were filtered out.
86    pub fn recount_bytes(&mut self) {
87        self.total_bytes = self.files.iter().map(|f| f.size).sum();
88    }
89}

97    /// A job for `src` → `/dst/<src>` with the given size and inode.
98    pub(crate) fn job_at(src: &str, size: u64, ino: u64) -> FileJob {
99        FileJob {
100            src: PathBuf::from(src),
101            dst: PathBuf::from("/dst").join(src.trim_start_matches('/')),
102            size,
103            ino,
104            meta: FileMeta {
105                permissions: Permissions::from_mode(0o644),
106                modified: FileTime::zero(),
107                accessed: FileTime::zero(),
108            },
109        }
110    }
111
112    /// A job that only matters for its size.
113    pub(crate) fn job(size: u64) -> FileJob {
114        job_at("/s", size, 0)
115    }
