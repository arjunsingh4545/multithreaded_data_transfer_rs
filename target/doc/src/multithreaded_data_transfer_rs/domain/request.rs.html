3use std::path::PathBuf;
4
5use super::plan::Tuning;
6use super::topology::KindOverrides;
7
8/// What to do with files that already exist at the destination.
9#[derive(Debug, Clone, Copy, PartialEq, Eq, Default)]
10pub enum ExistingPolicy {
11    /// Replace them.
12    #[default]
13    Overwrite,
14    /// Leave them untouched.
15    Skip,
16    /// Skip only files whose size and modification time already match (resume).
17    Update,
18    /// Abort before copying anything.
19    Fail,
20}
21
22/// A complete description of one copy.
23#[derive(Debug, Clone, Default)]
24pub struct CopyRequest {
25    /// Source directory.
26    pub src: PathBuf,
27    /// Destination directory (created if missing).
28    pub dst: PathBuf,
29    /// Strategy and performance knobs.
30    pub tuning: Tuning,
31    /// Disk-kind overrides.
32    pub overrides: KindOverrides,
33    /// Handling of existing destination files.
34    pub existing: ExistingPolicy,
35}
36
37impl CopyRequest {
38    /// A request with default tuning.
39    pub fn new(src: impl Into<PathBuf>, dst: impl Into<PathBuf>) -> Self {
40        Self {
41            src: src.into(),
42            dst: dst.into(),
43            ..Self::default()
44        }
45    }
46
47    /// Sets the tuning.
48    #[must_use]
49    pub fn with_tuning(mut self, tuning: Tuning) -> Self {
50        self.tuning = tuning;
51        self
52    }
53
54    /// Sets the disk-kind overrides.
55    #[must_use]
56    pub fn with_overrides(mut self, overrides: KindOverrides) -> Self {
57        self.overrides = overrides;
58        self
59    }
60
61    /// Sets the existing-file policy.
62    #[must_use]
63    pub fn with_existing(mut self, existing: ExistingPolicy) -> Self {
64        self.existing = existing;
65        self
66    }
67}
