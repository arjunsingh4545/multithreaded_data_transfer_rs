10/// What kind of storage backs a path.
11#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash, Default)]
12pub enum DiskKind {
13    /// Solid-state (`NVMe`, SATA SSD, RAM-backed…): seeks are cheap.
14    Ssd,
15    /// Rotational hard disk: seeks are expensive, sequential I/O is king.
16    Hdd,
17    /// Could not be determined (network FS, tmpfs, btrfs anonymous devices…).
18    #[default]
19    Unknown,
20}
21
22impl DiskKind {
23    /// `true` for spinning disks.
24    pub const fn is_rotational(self) -> bool {
25        matches!(self, Self::Hdd)
26    }
27
28    /// The kind used for planning: unknown storage is treated like an SSD,
29    /// which matches the tool's historical (fully parallel) behaviour.
30    #[must_use]
31    pub const fn effective(self) -> Self {
32        match self {
33            Self::Unknown => Self::Ssd,
34            known => known,
35        }
36    }
37}
38
39impl fmt::Display for DiskKind {
40    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
41        f.write_str(match self {
42            Self::Ssd => "SSD",
43            Self::Hdd => "HDD",
44            Self::Unknown => "unknown",
45        })
46    }
47}

49/// A sysfs `queue/rotational` value that is neither `0` nor `1`.
50#[derive(Debug, Clone, PartialEq, Eq, Error)]
51#[error("unexpected rotational flag `{0}`")]
52pub struct ParseDiskKindError(String);
53
54impl TryFrom<&str> for DiskKind {
55    type Error = ParseDiskKindError;
56
57    /// Parses a sysfs `queue/rotational` value.
58    fn try_from(flag: &str) -> Result<Self, Self::Error> {
59        match flag.trim() {
60            "0" => Ok(Self::Ssd),
61            "1" => Ok(Self::Hdd),
62            other => Err(ParseDiskKindError(other.to_owned())),
63        }
64    }
65}

67/// A Linux `major:minor` device number.
68#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash, Default)]
69pub struct DeviceId {
70    major: u32,
71    minor: u32,
72}
73
74impl DeviceId {
75    /// Builds an id from its parts.
76    pub const fn new(major: u32, minor: u32) -> Self {
77        Self { major, minor }
78    }
79
80    /// Major number (driver).
81    pub const fn major(self) -> u32 {
82        self.major
83    }
84
85    /// Minor number (instance / partition).
86    pub const fn minor(self) -> u32 {
87        self.minor
88    }
89}
90
91impl From<u64> for DeviceId {
92    /// Decodes a `st_dev` value using the glibc encoding from `<sys/sysmacros.h>`.
93    #[expect(
94        clippy::cast_possible_truncation,
95        reason = "both values are masked to 32 bits"
96    )]
97    fn from(dev: u64) -> Self {
98        let major = ((dev >> 8) & 0x0fff) | ((dev >> 32) & 0xffff_f000);
99        let minor = (dev & 0x00ff) | ((dev >> 12) & 0xffff_ff00);
100        Self::new(major as u32, minor as u32)
101    }
102}
103
104/// A string that is not of the form `major:minor`.
105#[derive(Debug, Clone, PartialEq, Eq, Error)]
106#[error("expected `major:minor`, got `{0}`")]
107pub struct ParseDeviceIdError(String);
108
109impl FromStr for DeviceId {
110    type Err = ParseDeviceIdError;
111
112    fn from_str(s: &str) -> Result<Self, Self::Err> {
113        let err = || ParseDeviceIdError(s.to_owned());
114        let (major, minor) = s.trim().split_once(':').ok_or_else(err)?;
115        Ok(Self::new(
116            major.parse().map_err(|_| err())?,
117            minor.parse().map_err(|_| err())?,
118        ))
119    }
120}
121
122impl fmt::Display for DeviceId {
123    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
124        write!(f, "{}:{}", self.major, self.minor)
125    }
126}

128/// A whole physical (or virtual) disk.
129#[derive(Debug, Clone, PartialEq, Eq)]
130pub struct DiskInfo {
131    /// Id of the whole disk, not the partition.
132    pub id: DeviceId,
133    /// Kernel name such as `sda` or `nvme0n1`.
134    pub name: String,
135    /// Storage kind.
136    pub kind: DiskKind,
137    /// Largest single request the device accepts (`queue/max_sectors_kb`).
138    pub max_request: Option<ByteSize>,
139}
140
141impl DiskInfo {
142    /// A disk with no request-size information.
143    pub fn new(id: DeviceId, name: impl Into<String>, kind: DiskKind) -> Self {
144        Self {
145            id,
146            name: name.into(),
147            kind,
148            max_request: None,
149        }
150    }
151
152    /// A device that could not be identified.
153    pub fn unknown(id: DeviceId) -> Self {
154        Self::new(id, "unknown", DiskKind::Unknown)
155    }
156
157    /// Replaces the kind (used for user overrides).
158    #[must_use]
159    pub fn with_kind(mut self, kind: DiskKind) -> Self {
160        self.kind = kind;
161        self
162    }
163}
