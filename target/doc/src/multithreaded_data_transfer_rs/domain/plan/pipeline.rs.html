1//! Planning for the reader → writer pipeline (HDD-friendly).
2
3use std::num::NonZeroUsize;
4
5use super::MIN_CHUNK;
6use super::tuning::{OrderKind, Tuning};
7use crate::domain::topology::TransferKind;
8use crate::util::units::ByteSize;
9
10/// Default I/O size per request: large enough to amortise a seek.
11pub const DEFAULT_CHUNK: ByteSize = ByteSize::mib(8);
12/// Default buffer memory between readers and writers.
13pub const DEFAULT_MEM: ByteSize = ByteSize::gib(1);
14/// Writeback window: dirty data is pushed to disk every this many bytes per file.
15pub const DEFAULT_FLUSH_WINDOW: ByteSize = ByteSize::mib(64);
16
17/// Parameters for pipeline mode.
18#[derive(Debug, Clone, Copy, PartialEq, Eq)]
19pub struct PipelinePlan {
20    /// Reader threads.
21    pub readers: NonZeroUsize,
22    /// Writer threads (each owns a lane of files).
23    pub writers: NonZeroUsize,
24    /// Bytes per read/write request.
25    pub chunk: ByteSize,
26    /// Number of `chunk`-sized buffers in flight.
27    pub buffers: NonZeroUsize,
28    /// Writeback window per file.
29    pub flush_every: ByteSize,
30    /// Copy order.
31    pub order: OrderKind,
32}
33
34impl PipelinePlan {
35    /// Upper bound on buffer memory.
36    pub fn memory(&self) -> ByteSize {
37        ByteSize::new(self.chunk.get().saturating_mul(self.buffers.get() as u64))
38    }
39}
40
41/// Per-transfer defaults before user overrides.
42struct Profile {
43    readers: usize,
44    writers: usize,
45    order: OrderKind,
46    mem_factor: u64,
47}
48
49fn profile(transfer: TransferKind, cores: usize) -> Profile {
50    let wide = cores.clamp(1, 8);
51    let (readers, writers, order, mem_factor) = match transfer {
52        TransferKind::HddToHdd => (1, 1, OrderKind::Physical, 1),
53        // One disk: bigger batches mean fewer head switches between the two areas.
54        TransferKind::SameHdd => (1, 1, OrderKind::Physical, 4),
55        // Cheap parallel reads keep a single sequential writer saturated.
56        TransferKind::SsdToHdd => (wide, 1, OrderKind::Path, 1),
57        TransferKind::HddToSsd => (1, wide.min(4), OrderKind::Physical, 1),
58        TransferKind::SsdToSsd => (wide, wide, OrderKind::Largest, 1),
59    };
60    Profile {
61        readers,
62        writers,
63        order,
64        mem_factor,
65    }
66}
67
68/// Builds a pipeline plan for `transfer`, applying any user overrides.
69pub fn pipeline_plan(transfer: TransferKind, tuning: &Tuning, cores: NonZeroUsize) -> PipelinePlan {
70    let profile = profile(transfer, cores.get());
71    let nz = |n: usize| NonZeroUsize::new(n).unwrap_or(NonZeroUsize::MIN);
72
73    let readers = tuning.readers.unwrap_or_else(|| nz(profile.readers));
74    let writers = tuning.writers.unwrap_or_else(|| nz(profile.writers));
75    let chunk = tuning.chunk.unwrap_or(DEFAULT_CHUNK).max(MIN_CHUNK);
76    let mem = tuning
77        .mem
78        .unwrap_or_else(|| ByteSize::new(DEFAULT_MEM.get().saturating_mul(profile.mem_factor)));
79
80    // Every thread must be able to hold a couple of buffers or the pipeline stalls.
81    let min_buffers = 2 * (readers.get() + writers.get());
82    let buffers = (mem.get() / chunk.get())
83        .try_into()
84        .unwrap_or(usize::MAX)
85        .max(min_buffers);
86
87    PipelinePlan {
88        readers,
89        writers,
90        chunk,
91        buffers: nz(buffers),
92        flush_every: DEFAULT_FLUSH_WINDOW.max(chunk),
93        order: tuning.order.unwrap_or(profile.order),
94    }
95}

97#[cfg(test)]
98mod tests {
99    use super::*;
100
101    const EIGHT: NonZeroUsize = NonZeroUsize::new(8).unwrap();
102
103    fn auto(transfer: TransferKind) -> PipelinePlan {
104        pipeline_plan(transfer, &Tuning::default(), EIGHT)
105    }
106
107    #[test]
108    fn hdd_to_hdd_uses_one_reader_and_one_writer() {
109        let plan = auto(TransferKind::HddToHdd);
110        assert_eq!((plan.readers.get(), plan.writers.get()), (1, 1));
111    }
112
113    #[test]
114    fn ssd_to_hdd_fans_in_to_one_writer() {
115        let plan = auto(TransferKind::SsdToHdd);
116        assert_eq!((plan.readers.get(), plan.writers.get()), (8, 1));
117    }
118
119    #[test]
120    fn hdd_source_reads_in_physical_order() {
121        assert_eq!(auto(TransferKind::HddToSsd).order, OrderKind::Physical);
122    }
123
124    #[test]
125    fn same_disk_gets_four_times_the_memory() {
126        assert_eq!(auto(TransferKind::SameHdd).memory(), ByteSize::gib(4));
127    }
128
129    #[test]
130    fn default_memory_is_one_gib() {
131        assert_eq!(auto(TransferKind::HddToHdd).memory(), DEFAULT_MEM);
132    }
133
134    #[test]
135    fn buffers_at_least_twice_thread_count() {
136        let tuning = Tuning {
137            mem: Some(ByteSize::mib(8)),
138            ..Tuning::default()
139        };
140        let plan = pipeline_plan(TransferKind::SsdToHdd, &tuning, EIGHT);
141        assert_eq!(plan.buffers.get(), 2 * (8 + 1));
142    }
143
144    #[test]
145    fn cli_readers_override_auto() {
146        let tuning = Tuning {
147            readers: NonZeroUsize::new(3),
148            ..Tuning::default()
149        };
150        let plan = pipeline_plan(TransferKind::HddToHdd, &tuning, EIGHT);
151        assert_eq!(plan.readers.get(), 3);
152    }
153
154    #[test]
155    fn chunk_override_applies() {
156        let tuning = Tuning {
157            chunk: Some(ByteSize::mib(16)),
158            ..Tuning::default()
159        };
160        assert_eq!(
161            pipeline_plan(TransferKind::HddToHdd, &tuning, EIGHT).chunk,
162            ByteSize::mib(16)
163        );
164    }
165
166    #[test]
167    fn flush_window_never_smaller_than_chunk() {
168        let tuning = Tuning {
169            chunk: Some(ByteSize::mib(128)),
170            ..Tuning::default()
171        };
172        let plan = pipeline_plan(TransferKind::HddToHdd, &tuning, EIGHT);
173        assert_eq!(plan.flush_every, ByteSize::mib(128));
174    }
175}
