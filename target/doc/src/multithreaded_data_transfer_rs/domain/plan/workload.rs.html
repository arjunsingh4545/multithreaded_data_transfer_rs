1//! Aggregate workload numbers the planner reasons about.
2
3use crate::domain::job::ScanManifest;
4use crate::util::units::ByteSize;
5
6/// File count and volume of a copy.
7#[derive(Debug, Clone, Copy, Default, PartialEq, Eq)]
8pub struct WorkloadStats {
9    /// Number of regular files.
10    pub files: usize,
11    /// Total bytes.
12    pub total: ByteSize,
13}
14
15impl WorkloadStats {
16    /// Average file size (zero for an empty workload).
17    pub fn average(&self) -> ByteSize {
18        ByteSize::new(self.total.get() / (self.files.max(1) as u64))
19    }
20}
21
22impl From<&ScanManifest> for WorkloadStats {
23    fn from(manifest: &ScanManifest) -> Self {
24        Self {
25            files: manifest.files.len(),
26            total: ByteSize::new(manifest.total_bytes),
27        }
28    }
29}

31#[cfg(test)]
32mod tests {

34
35    #[test]
36    fn average_handles_zero_files() {
37        assert_eq!(WorkloadStats::default().average(), ByteSize::new(0));
38    }
39
40    #[test]
41    fn average_divides_total_by_count() {
42        let stats = WorkloadStats {
43            files: 4,
44            total: ByteSize::kib(4),
45        };
46        assert_eq!(stats.average(), ByteSize::kib(1));
47    }
