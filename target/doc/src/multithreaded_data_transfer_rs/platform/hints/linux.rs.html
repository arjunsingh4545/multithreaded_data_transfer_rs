1//! Linux implementation: `posix_fadvise`, `fallocate`, `sync_file_range`.
2
3use std::fs::File;
4use std::io;
5use std::ops::Range;
6use std::os::fd::AsRawFd;
7
8use super::IoHints;
9use crate::platform::ffi::{cvt, cvt_errno, to_off};
10
11/// Hints backed by real Linux syscalls.
12#[derive(Debug, Clone, Copy, Default)]
13pub struct LinuxHints;
14
15fn fadvise(file: &File, range: &Range<u64>, advice: libc::c_int) -> io::Result<()> {
16    let offset = to_off(range.start)?;
17    let len = to_off(range.end.saturating_sub(range.start))?;
18    // SAFETY: the fd is owned by `file`, which stays borrowed (and therefore open)
19    // for the duration of the call. posix_fadvise does not touch user memory.
20    cvt_errno(unsafe { libc::posix_fadvise(file.as_raw_fd(), offset, len, advice) })
21}
22
23fn sync_range(file: &File, range: &Range<u64>, flags: libc::c_uint) -> io::Result<()> {
24    let offset = to_off(range.start)?;
25    let len = to_off(range.end.saturating_sub(range.start))?;
26    // SAFETY: as above — `file` keeps the fd valid; no user memory is passed.
27    cvt(unsafe { libc::sync_file_range(file.as_raw_fd(), offset, len, flags) }).map(drop)
28}
29
30impl IoHints for LinuxHints {
31    fn advise_sequential(&self, file: &File) -> io::Result<()> {
32        // A zero length means "to the end of the file".
33        fadvise(file, &(0..0), libc::POSIX_FADV_SEQUENTIAL)
34    }
35
36    fn preallocate(&self, file: &File, len: u64) -> io::Result<()> {
37        if len == 0 {
38            return Ok(());
39        }
40        let len = to_off(len)?;
41        // SAFETY: `file` keeps the fd valid; mode 0 allocates and extends the file size,
42        // which `PendingFile::finish` later trims to the bytes actually written.
43        cvt(unsafe { libc::fallocate(file.as_raw_fd(), 0, 0, len) }).map(drop)
44    }
45
46    fn start_writeback(&self, file: &File, range: Range<u64>) -> io::Result<()> {
47        sync_range(file, &range, libc::SYNC_FILE_RANGE_WRITE)
48    }
49
50    fn settle_and_evict(&self, file: &File, range: Range<u64>) -> io::Result<()> {
51        sync_range(
52            file,
53            &range,
54            libc::SYNC_FILE_RANGE_WAIT_BEFORE
55                | libc::SYNC_FILE_RANGE_WRITE
56                | libc::SYNC_FILE_RANGE_WAIT_AFTER,
57        )?;
58        fadvise(file, &range, libc::POSIX_FADV_DONTNEED)
59    }
60
61    fn evict(&self, file: &File, range: Range<u64>) -> io::Result<()> {
62        fadvise(file, &range, libc::POSIX_FADV_DONTNEED)
63    }
64}

66#[cfg(test)]
67mod tests {
68    use super::*;
69    use crate::platform::ffi::is_unsupported;
70    use std::os::unix::fs::FileExt;
71
72    /// Hints are best-effort; on tmpfs some calls are unsupported, which is fine.
73    fn ok_or_unsupported(result: &io::Result<()>) -> bool {
74        result.as_ref().map_or_else(is_unsupported, |()| true)
75    }
76
77    #[test]
78    fn advise_sequential_succeeds_on_tempfile() {
79        let file = tempfile::tempfile().unwrap();
80        assert!(ok_or_unsupported(&LinuxHints.advise_sequential(&file)));
81    }
82
83    #[test]
84    fn preallocate_extends_file() {
85        let file = tempfile::tempfile().unwrap();
86        let result = LinuxHints.preallocate(&file, 1 << 20);
87        if result.is_ok() {
88            assert_eq!(file.metadata().unwrap().len(), 1 << 20);
89        } else {
90            assert!(ok_or_unsupported(&result));
91        }
92    }
93
94    #[test]
95    fn writeback_cycle_succeeds() {
96        let file = tempfile::tempfile().unwrap();
97        file.write_all_at(&[7u8; 8192], 0).unwrap();
98        assert!(ok_or_unsupported(
99            &LinuxHints.start_writeback(&file, 0..8192)
100        ));
101        assert!(ok_or_unsupported(
102            &LinuxHints.settle_and_evict(&file, 0..8192)
103        ));
104    }
105}
